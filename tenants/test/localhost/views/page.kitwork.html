app.kitwork.js
import { router, database } from "kitwork"

const db = database.connection()

// 1. Zero-Allocation REST API & Router Caching
router.get("/api/v1/orders/:id")
    .cache("30s")
    .handle((req, res) => {
        const order = db.table("orders").find(req.params("id"))
        if (!order) return res.status(404).json({ error: "Not found" })
        return res.json({ success: true, data: order })
    })

// 1. Zero-Allocation JSON API
router.get("/api/hello").handle((req, res) => {
    return res.json({
        status: "active",
        engine: "Kitwork Sovereign VM",
        timestamp: Date().toISOString()
    })
})

// 2. Disk-backed Router cache
router.get("/api/cached")
    .static("10s")
    .handle((req, res) => {
        return res.text("Cached response")
    })
const db = database.connection()

// Fluent Zero-Allocation ORM
router.get("/test-query").handle((req, res) => {
    return res.json({
        find: db.table("user").find("username", "grace"),
        first: db.table("user").first(),
        count: db.table("user").count()
    })
})
// Compliant Napas VietQR SVGs
router.get("/qrcode/napas").handle((req, res) => {
    const payment = napas
        .bank(req.query("bank").text(), req.query("account").text())
        .amount(req.query("amount").text())
        .info(req.query("desc").text())

    const svg = qrcode
        .napas(payment)
        .template("circular")
        .cell({ color: "#0f172a" })
        .svg()

    return res.svg(svg)
})
// Headless Web Screenshot (chromedp)
router.get("/screenshot").handle((req, res) => {
    const url = req.query("url").text()
    const png = chromedp.capture(url, {
        width: 1280,
        height: 720
    })
    return res.image(png)
})
// Spawn background goroutine parallel worker
router.get("/background").handle((req, res) => {
    go(() => {
        log.Print("Goroutine is running...")
        // Non-blocking database sync worker
        log.Print("Goroutine finished!")
    })

    return res.text("Background task started!")
})
